/** * 申し込み管理システム（moushikomi-59749）Firestore 自動バックアップ GAS * - Googleドライブの指定フォルダへ JSON バックアップを保存（復元用） * - 紐づいたGoogleスプレッドシートへ「申込一覧」「フォルダ一覧」を書き出し */ const FIRESTORE_PROJECT_ID = 'moushikomi-59749'; const BACKUP_DRIVE_FOLDER_NAME = '申し込み管理_自動バックアップ'; const BACKUP_DRIVE_FOLDER_ID = ''; // 特定の既存フォルダIDに固定したい場合はここに入力（例: '1AbCdEfGhIj...'） const KEEP_DAYS = 30; // この日数を過ぎたJSONバックアップをゴミ箱へ移動（0にすると削除しない） /** * 毎日自動実行（トリガー）用のメイン関数 */ function runDailyFirestoreBackup() { const now = new Date(); const timestampLabel = Utilities.formatDate(now, 'Asia/Tokyo', 'yyyyMMdd_HHmm'); const displayTime = Utilities.formatDate(now, 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss'); // 1. Firestore から events と submissions を全件取得 const events = fetchAllDocuments('events'); const submissions = fetchAllDocuments('submissions'); // 2. 復元用 JSON ペイロード作成 const backupPayload = { appName: 'moushikomi-kanri', projectId: FIRESTORE_PROJECT_ID, exportedAt: now.toISOString(), exportedAtJst: displayTime, counts: { events: events.length, submissions: submissions.length }, events: events, submissions: submissions }; const jsonString = JSON.stringify(backupPayload, null, 2); const fileName = `申し込みバックアップ_${timestampLabel}.json`; // 3. Googleドライブに JSON 保存＆古いファイルの整理 const folder = getTargetBackupFolder(); const savedFile = saveJsonToDrive(folder, fileName, jsonString); if (KEEP_DAYS > 0) { cleanupOldBackups(folder, KEEP_DAYS); } // 4. スプレッドシートに一覧を書き出し writeToSpreadsheet(events, submissions, displayTime); Logger.log( `バックアップ完了: ${savedFile.getName()} (フォルダ: ${events.length}件, 申込: ${submissions.length}件)` ); } /** * Firestore REST API から指定コレクションの全ドキュメントを取得 */ function fetchAllDocuments(collectionName) { const results = []; let pageToken = null; do { const baseUrl = `https://firestore.googleapis.com/v1/projects/${encodeURIComponent(FIRESTORE_PROJECT_ID)}` + `/databases/(default)/documents/${encodeURIComponent(collectionName)}?pageSize=300`; const url = pageToken ? `${baseUrl}&pageToken=${encodeURIComponent(pageToken)}` : baseUrl; const response = UrlFetchApp.fetch(url, { method: 'get', muteHttpExceptions: true }); const statusCode = response.getResponseCode(); const bodyText = response.getContentText(); if (statusCode < 200 || statusCode >= 300) { throw new Error( `Firestore取得エラー (${collectionName} / HTTP ${statusCode}): ${bodyText}` ); } const data = JSON.parse(bodyText); const documents = Array.isArray(data.documents) ? data.documents : []; documents.forEach((doc) => { const parts = (doc.name || '').split('/'); const docId = parts[parts.length - 1] || ''; const fieldsObj = decodeFirestoreFields(doc.fields || {}); results.push({ id: docId, ...fieldsObj }); }); pageToken = data.nextPageToken || null; } while (pageToken); return results; } /** * Firestore REST API の fields 構造を通常の JS オブジェクトに変換 */ function decodeFirestoreFields(fields) { const obj = {}; Object.keys(fields).forEach((key) => { obj[key] = decodeFirestoreValue(fields[key]); }); return obj; } function decodeFirestoreValue(val) { if (!val || typeof val !== 'object') return null; if ('stringValue' in val) return val.stringValue; if ('booleanValue' in val) return val.booleanValue; if ('integerValue' in val) return Number(val.integerValue); if ('doubleValue' in val) return Number(val.doubleValue); if ('nullValue' in val) return null; if ('timestampValue' in val) return val.timestampValue; if ('bytesValue' in val) return val.bytesValue; if ('geoPointValue' in val) return val.geoPointValue; if ('referenceValue' in val) return val.referenceValue; if ('arrayValue' in val) { const values = (val.arrayValue && Array.isArray(val.arrayValue.values)) ? val.arrayValue.values : []; return values.map((item) => decodeFirestoreValue(item)); } if ('mapValue' in val) { const mapFields = (val.mapValue && val.mapValue.fields) ? val.mapValue.fields : {}; return decodeFirestoreFields(mapFields); } return null; } /** * 保存先Googleドライブフォルダを取得（ID指定優先、なければ名前で検索または作成） */ function getTargetBackupFolder() { if (BACKUP_DRIVE_FOLDER_ID && BACKUP_DRIVE_FOLDER_ID.trim() !== '') { return DriveApp.getFolderById(BACKUP_DRIVE_FOLDER_ID.trim()); } const folders = DriveApp.getFoldersByName(BACKUP_DRIVE_FOLDER_NAME); if (folders.hasNext()) { return folders.next(); } return DriveApp.createFolder(BACKUP_DRIVE_FOLDER_NAME); } /** * JSONファイルをGoogleドライブに保存 */ function saveJsonToDrive(folder, fileName, jsonString) { const blob = Utilities.newBlob(jsonString, 'application/json', fileName); return folder.createFile(blob); } /** * 指定日数より古いバックアップファイルをゴミ箱へ移動 */ function cleanupOldBackups(folder, keepDays) { const thresholdMs = Date.now() - keepDays * 24 * 60 * 60 * 1000; const files = folder.getFiles(); while (files.hasNext()) { const file = files.next(); const createdTime = file.getDateCreated().getTime(); if (createdTime < thresholdMs) { file.setTrashed(true); } } } /** * スプレッドシートに「申込一覧」と「フォルダ一覧」を出力 */ function writeToSpreadsheet(events, submissions, stampText) { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss) { throw new Error('紐づいたスプレッドシートが見つかりません。スプレッドシートの「拡張機能＞Apps Script」から開いてください。'); } // イベントID -> 名称のマップを作成 const eventNameMap = {}; events.forEach((ev) => { eventNameMap[ev.id] = ev.name || ev.title || ''; }); // 1. 申込一覧シート let subSheet = ss.getSheetByName('申込一覧'); if (!subSheet) { subSheet = ss.insertSheet('申込一覧'); } subSheet.clearContents(); const sortedSubmissions = submissions.slice().sort((a, b) => { const timeA = toEpochMs(a.createdAt); const timeB = toEpochMs(b.createdAt); return timeB - timeA; }); const subRows = [ ['最終更新日時', stampText, '', '', '', '', '', '', '', '', '', ''], [ 'No', '申し込みフォルダ', '氏名', 'フリガナ', '学年', '性別', '生年月日', '種目', '住所', '連絡先', '申込日時', 'データID' ] ]; sortedSubmissions.forEach((item, idx) => { const folderLabel = item.eventName || eventNameMap[item.eventId] || item.eventId || ''; const createdAtText = formatTimestampJst(item.createdAt); subRows.push([ idx + 1, folderLabel, item.name || '', item.kana || item.furigana || '', item.grade || '', item.gender || item.sex || '', item.birthDate || item.birthday || '', item.category || item.eventType || item.type || '', item.address || '', item.phone || item.tel || item.contact || '', createdAtText, item.id || '' ]); }); subSheet.getRange(1, 1, subRows.length, 12).setValues(subRows); // 2. フォルダ一覧シート let evSheet = ss.getSheetByName('フォルダ一覧'); if (!evSheet) { evSheet = ss.insertSheet('フォルダ一覧'); } evSheet.clearContents(); const evRows = [ ['最終更新日時', stampText, '', ''], ['No', '申し込み名称', 'フォルダID', '作成日時'] ]; events.forEach((ev, idx) => { evRows.push([ idx + 1, ev.name || ev.title || '', ev.id || '', formatTimestampJst(ev.createdAt) ]); }); evSheet.getRange(1, 1, evRows.length, 4).setValues(evRows); } function toEpochMs(val) { if (!val) return 0; if (typeof val === 'number') return val; const parsed = Date.parse(val); return isNaN(parsed) ? 0 : parsed; } function formatTimestampJst(val) { const ms = toEpochMs(val); if (!ms) return ''; return Utilities.formatDate(new Date(ms), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm'); }
